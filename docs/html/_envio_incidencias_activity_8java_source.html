32 * Esta actividad se encarga de:

45public class EnvioIncidenciasActivity extends AppCompatActivity {
46 /**
47 * @brief Método de inicialización de la actividad.
48 *
49 * Se encarga de:
50 * - Vincular las vistas del layout.
51 * - Recuperar datos del Intent (SENSOR_NAME, UBICACION, ULTIMA_CONEXION).
52 * - Autocompletar los campos de texto con un mensaje preformateado.
53 * - Configurar los listeners para los botones de Aceptar y Cancelar.
54 *
55 * @param savedInstanceState Estado guardado de la aplicación (si existe).

57 @Override
58 protected void onCreate(Bundle savedInstanceState) {
59 super.onCreate(savedInstanceState);
60 setContentView(R.layout.envio_incidencias);
61
62 // Encontrar las vistas
63 // Campo de texto para el título de la incidencia.
64 EditText tituloEditText = findViewById(R.id.editText_tituloIncidencia);
65 // Campo de texto para el mensaje detallado de la incidencia.
66 EditText mensajeEditText = findViewById(R.id.editText_mensaje);
67 // Botón para enviar la incidencia.
68 Button acceptButton = findViewById(R.id.button_aceptar);
69 // Botón para cancelar y cerrar la actividad.
70 Button cancelButton = findViewById(R.id.button_cancelar);
71
72 // --- Lógica para rellenar los datos automáticamente ---
73 Intent intent = getIntent();
74 // Nombre o código del sensor recibido de la actividad anterior.
75 String sensorName = intent.getStringExtra("SENSOR_NAME");
76 // Ubicación actual del sensor (coordenadas convertidas a dirección).
77 String ubicacion = intent.getStringExtra("UBICACION");
78 // Cadena de texto que indica la hora de la última conexión.
79 String ultimaConexion = intent.getStringExtra("ULTIMA_CONEXION");
80
81 if (ultimaConexion != null) {
82 ultimaConexion = ultimaConexion.replace("Última conex. ", "");
83 }
84
85 // --- Lógica para rellenar los campos de texto ---
86 // Rellenamos automaticamente el asunto y el mensaje
87 String titulo = String.format("AVISO: Sensor %s Desconectado", sensorName);
88 String mensaje = String.format(Locale.getDefault(),
89 "El sensor %s de la zona %s ha dejado de funcionar. La última lectura se recibió a las %s. Por favor, compruebe la conexión o si existe algún problema con el sensor.",
90 sensorName, ubicacion, ultimaConexion);
91
92 tituloEditText.setText(titulo);
93 mensajeEditText.setText(mensaje);
94
95 // --- Lógica de los botones ---
96 // Listener para el botón Cancelar. Cierra la actividad.
97 cancelButton.setOnClickListener(v -> {
98 finish();
99 });
100
101 //Listener para el botón Aceptar (Enviar).
102 // Recoge los datos, crea un mapa de incidencia y lo sube a Firebase Firestore
103 acceptButton.setOnClickListener(v -> {
104 String tituloIncidencia = tituloEditText.getText().toString();
105 String mensajeIncidencia = mensajeEditText.getText().toString();
106
107 // 0. Validación básica antes de enviar nada
108 if (tituloIncidencia.isEmpty() || mensajeIncidencia.isEmpty()) {
109 Toast.makeText(this, "Por favor, rellena título y mensaje", Toast.LENGTH_SHORT).show();
110 return;
111 }
112
113 // ------ Implementación Envío de Correo (JavaMail) -----------------
114 // Dirección de correo del administrador que recibirá la alerta. */
115 String emailDestino = "sandralovesel@gmail.com";
116
117 // Asunto del correo concatenando el prefijo fijo y el título del usuario. */
118 String asuntoCorreo = tituloIncidencia;
119
120 // Instancia de la clase asíncrona encargada de la conexión SMTP con el 'Robot'. */
121 JavaMailAPI mailSender = new JavaMailAPI(this, emailDestino, asuntoCorreo, mensajeIncidencia);
122
123 // Ejecuta el envío del correo electrónico en segundo plano. */
124 mailSender.execute();
125 // ------------------------------------------------------------------
126
127
128 // ------ Implementación Firebase ---------------------------
129 FirebaseFirestore db = FirebaseFirestore.getInstance();
130 Map<String, Object> incidencia = new HashMap<>();
131
132 // Código o nombre del sensor afectado (asegúrate de tener esta variable definida). */
133 incidencia.put("sensor_id", sensorName);
134 // Título de la incidencia (manual o precargado). */
135 incidencia.put("titulo", tituloIncidencia);
136 // Contenido detallado de la incidencia (manual o precargado). */
137 incidencia.put("mensaje", mensajeIncidencia);
138 // Ubicación registrada del sensor en el momento de la incidencia. */
139 incidencia.put("ubicacion", ubicacion);
140 // Estado inicial de la incidencia. */
141 incidencia.put("estado", "PENDIENTE");
142 // Indicador de si la incidencia ha sido resuelta. */
143 incidencia.put("resuelta", false);
144 // Timestamp del servidor para registrar la fecha de envío exacta. */
145 incidencia.put("fecha", com.google.firebase.firestore.FieldValue.serverTimestamp());
146 // ------------------------------------------
147
148 // Sube el documento a la colección 'incidencias'
149 db.collection("incidencias").add(incidencia)
150 .addOnSuccessListener(documentReference -> {
151 //PARA EL CAMBIO DE BOTON DE INCIDENCIAS
152 setResult(RESULT_OK);
153 // Muestra un diálogo de éxito y cierra la actividad al aceptar. */
154 new androidx.appcompat.app.AlertDialog.Builder(this)
155 .setTitle("Enviado")
156 .setMessage("Incidencia registrada en el sistema y correo enviado al administrador.")
157 .setPositiveButton("Aceptar", (dialog, which) -> {
158 finish();
159 })
160 .setCancelable(false)
161 .show();
162 })
163 .addOnFailureListener(e -> {
164 // Manejo de errores en caso de fallo de conexión con Firebase. */
165 new androidx.appcompat.app.AlertDialog.Builder(this)
166 .setTitle("Error de Envío")
167 .setMessage("No se pudo guardar la incidencia en la base de datos. Error: " + e.getMessage())
168 .setPositiveButton("Aceptar", null)
169 .show();
170 });
171 });
172 }
