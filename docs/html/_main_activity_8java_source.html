51 * @details
52 * Esta clase gestiona el acceso seguro a la aplicación. Implementa tres métodos de entrada:
53 * 1. **Manual:** El usuario introduce el código alfanumérico del sensor.
54 * 2. **QR:** Escaneo mediante la cámara usando la librería ZXing.
55 * 3. **Biométrico:** Acceso rápido mediante huella dactilar si ya hubo un inicio de sesión previo exitoso.

223 /**
224 * @brief Muestra el diálogo de autenticación biométrica (huella dactilar).
225 * Si la autenticación es exitosa, intenta recuperar el último ID de sensor guardado para iniciar la sesión rápida.
226 * () -> mostrarAutenticacionBiometrica() -> ()
227 *
228 * Lógica:
229 * - Si la huella es válida -> Recupera `LAST_SENSOR_ID` de SharedPreferences.
230 * - Si hay un ID guardado -> Inicia sesión automáticamente (@ref iniciarSesionExitosa).
231 * - Si no hay ID -> Mantiene al usuario en la pantalla para login manual.
232 */
233 private void mostrarAutenticacionBiometrica() {
234 BiometricPrompt.PromptInfo promptInfo = new BiometricPrompt.PromptInfo.Builder()
235 .setTitle("Acceso Biométrico")
236 .setSubtitle("Usa tu huella dactilar para iniciar sesión")
237 .setNegativeButtonText("Usar código de sensor") // Opción para volver al login normal
238 .setConfirmationRequired(true)
239 .build();
240
241 BiometricPrompt biometricPrompt = new BiometricPrompt(
242 MainActivity.this,
243 ContextCompat.getMainExecutor(this),
244 new BiometricPrompt.AuthenticationCallback() {
245 @Override
246 public void onAuthenticationSucceeded(@NonNull BiometricPrompt.AuthenticationResult result) {
247 super.onAuthenticationSucceeded(result);
248 // 1. Recuperar el último ID guardado
249 SharedPreferences prefs = getSharedPreferences("AppPrefs", MODE_PRIVATE);
250 String lastSensorId = prefs.getString("LAST_SENSOR_ID", null);
251
252 if (lastSensorId != null) {
253 // 2. Si se encuentra un ID real, iniciar la sesión con ese ID.
254 Log.d("BIOMETRIA", "Acceso biométrico exitoso. Usando ID guardado: " + lastSensorId);
255 iniciarSesionExitosa(lastSensorId); // ¡Usamos el ID real aquí!
256 } else {
257 // 3. Caso de primer uso: No hay ID guardado.
258 Toast.makeText(MainActivity.this, "Inicie sesión con un código primero para habilitar el acceso rápido.", Toast.LENGTH_LONG).show();
259 // Nota: Mantenemos el usuario en la pantalla de login.
260 }
261 }
262
263 @Override
264 public void onAuthenticationError(int errorCode, @NonNull CharSequence errString) {
265 super.onAuthenticationError(errorCode, errString);
266 // Error
267 Toast.makeText(MainActivity.this, "Error de autenticación: " + errString, Toast.LENGTH_SHORT).show();
268 }
269 });
270
271 biometricPrompt.authenticate(promptInfo);
272 }

274 /**
275 * @brief Guarda el código del sensor en SharedPreferences e inicia la \ref SesionSensorActivity.
276 * Esta función se llama tras un login exitoso (manual o biométrico).
277 * (code:String) -> iniciarSesionExitosa() -> ()
278 *
279 * @param code El código de sensor verificado.
280 */
281 private void iniciarSesionExitosa(String code) {
282 // Esta acción se realiza siempre, sobreescribiendo el ID con el mismo ID, lo cual es seguro.
283 SharedPreferences prefs = getSharedPreferences("AppPrefs", MODE_PRIVATE);
284 prefs.edit().putString("LAST_SENSOR_ID", code).apply();
285 Log.d("BIOMETRIA", "ID de sensor guardado/actualizado: " + code);
286
287 // Inicio de actividad
288 Intent intent = new Intent(MainActivity.this, SesionSensorActivity.class);
289 intent.putExtra("SENSOR_CODE", code);
290 startActivity(intent);
291 finish();
292 }

298 /**
299 * @brief Verifica asíncronamente si el código del sensor existe como documento en la colección 'sensores' de Firebase Firestore.
300 * (sensorCode:String) -> checkSensorCodeInDatabase() -> ()
301 *
302 * Busca el documento con ID `sensorCode` en la colección `sensores`.
303 * - Si existe: Llama a @ref iniciarSesionExitosa.
304 * - Si no existe o error: Muestra AlertDialog.
305 *
306 * @param sensorCode El código de sensor ingresado por el usuario.
307 */
308 private void checkSensorCodeInDatabase(String sensorCode) {
309 Toast.makeText(this, "Verificando código en Firestore...", Toast.LENGTH_SHORT).show();
310
311 // Referencia específica al documento del sensor: coleccion(sensores)/documento(código)
312 db.collection("sensores").document(sensorCode)
313 .get() // Consulta de un solo evento
314 .addOnCompleteListener(new OnCompleteListener<DocumentSnapshot>() {
315 @Override
316 public void onComplete(@NonNull Task<DocumentSnapshot> task) {
317 if (task.isSuccessful()) {
318 DocumentSnapshot document = task.getResult();
319 // Diseño Lógico: Validación de Existencia
320 if (document.exists()) {
321 // Éxito: El documento existe en la colección 'sensores'.
322 Log.d("Firestore_LOGIN", "Sensor encontrado: " + sensorCode);
323 Toast.makeText(MainActivity.this, "Conexión exitosa con el sensor.", Toast.LENGTH_SHORT).show();
324 iniciarSesionExitosa(sensorCode);
325 } else {
326 // Fallo 1: El código NO existe
327 Log.e("Firestore_LOGIN", "Código NO encontrado: " + sensorCode);
328 new AlertDialog.Builder(MainActivity.this)
329 .setTitle("Error de Vinculación")
330 .setMessage("El código de sensor '" + sensorCode + "' no es válido o no está registrado.")
331 .setPositiveButton("Aceptar", null)
332 .show();
333 }
334 } else {
335 // Fallo 2: Error de conexión o tarea
336 Log.e("Firestore_LOGIN", "Fallo al consultar Firestore: ", task.getException());
337 new AlertDialog.Builder(MainActivity.this)
338 .setTitle("Error de Conexión")
339 .setMessage("Fallo al consultar la BD: " + task.getException().getMessage())
340 .setPositiveButton("Aceptar", null)
341 .show();
342 }
343 }
344 });
345 }
