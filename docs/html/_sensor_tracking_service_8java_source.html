358 /**
359 * @brief Procesa el resultado del escaneo BLE, filtra el dispositivo "rocio" y decodifica el payload.
360 * Actualiza el estado del DataHolder, verifica alertas y sube los datos a Firebase.
361 * (resultado:ScanResult) -> mostrarInformacionDispositivoBTLE() -> ()
362 *
363 * @details
364 * **Proceso de Decodificación:**
365 * 1. Filtra por nombre de dispositivo ("rocio").
366 * 2. Calcula RSSI suavizado usando Media Móvil Exponencial:
367 * \f$ RSSI_{smooth} = \alpha \cdot RSSI_{raw} + (1-\alpha) \cdot RSSI_{prev} \f$
368 * 3. Extrae Manufacturer Data (ID 0x004C).
369 * 4. Interpreta bytes mediante Bit Shifting para reconstruir enteros y floats.
370 * 5. Actualiza DataHolder y Firebase si los valores han cambiado.
371 *
372 *
373 * @param resultado El objeto ScanResult devuelto por el escáner BLE.
374 */
375 private void mostrarInformacionDispositivoBTLE(ScanResult resultado) {
376 // Comprobamos que el dispositivo es el correcto
377 BluetoothDevice device = resultado.getDevice();
378 if (device == null || device.getName() == null || !device.getName().equals("rocio")) return;
379
380 // Reiniciamos el watchdog
381 resetWatchdog();
382
383 // --- NUEVO: Actualizamos RSSI (Media Ponderada) ---
384 int rawRssi = resultado.getRssi();
385
386 if (smoothedRssi == -999.0f) {
387 smoothedRssi = rawRssi; // Primer valor, inicializamos
388 } else {
389 // Fórmula: NuevoPromedio = (alpha * NuevoValor) + ((1 - alpha) * PromedioAnterior)
390 smoothedRssi = (ALPHA_RSSI * rawRssi) + ((1.0f - ALPHA_RSSI) * smoothedRssi);
391 }
392
393 dataHolder.rssiData.postValue((int) smoothedRssi);
394 // ----------------------------------------
395
396 // Parseamos la trama
397 ScanRecord scanRecord = resultado.getScanRecord();
398 if (scanRecord == null) return;
399
400 // Extraemos los datos del beacon
401 byte[] payload = scanRecord.getManufacturerSpecificData(0x004C);
402 if (payload == null || payload.length != 9 || payload[0] != (byte) 0xAA) return;
403
404 // Parseamos los datos del beacon
405 float o3_ppm = ((payload[2] & 0xFF) << 8 | (payload[1] & 0xFF)) / 1000.0f;
406 float temperatura_c = ((payload[4] & 0xFF) << 8 | (payload[3] & 0xFF)) / 10.0f;
407 int co2_ppm = (payload[6] & 0xFF) << 8 | (payload[5] & 0xFF);
408 int bat_porc = (payload[8] & 0xFF) << 8 | (payload[7] & 0xFF);
409
410 // Comprobar si los valores de medición han cambiado.
411 if (temperatura_c == lastUpdatedTemp && o3_ppm == lastUpdatedOzono && co2_ppm == lastUpdatedCo2) {
412 dataHolder.bateriaData.postValue(bat_porc); // Actualizamos la batería siempre
413 return; // Si los datos principales no cambian, salimos.
414 }
415
416 // Si los datos SÍ han cambiado, actualizamos la memoria y procedemos.
417 Log.i(ETIQUETA_LOG, "¡Nuevos datos de sensor detectados! Actualizando UI y hora.");
418 lastUpdatedTemp = temperatura_c;
419 lastUpdatedOzono = o3_ppm;
420 lastUpdatedCo2 = co2_ppm;
421
422 // Comprobamos las alertas
423 checkAlerts(co2_ppm, o3_ppm, temperatura_c, bat_porc);
424
425 // Actualizamos la UI con los nuevos datos
426 // Obtenemos la hora actual para saber la ultima actualizacion de datos
427 String receptionTime = new SimpleDateFormat("HH:mm", Locale.getDefault()).format(new Date());
428 dataHolder.timeData.postValue("Última conex. " + receptionTime);
429 dataHolder.ozonoData.postValue(o3_ppm);
430 dataHolder.temperaturaData.postValue(temperatura_c);
431 dataHolder.co2Data.postValue(co2_ppm);
432 dataHolder.bateriaData.postValue(bat_porc);
433
434 // Lógica de subida a firestore
435 // Obtenemos los valores de contexto (que se actualizaron en los otros hilos/callbacks)
436 String ubicacion = dataHolder.locationData.getValue();
437 String estado = dataHolder.estadoData.getValue();
438
439 // Llamamos a la función de subida. Pasamos los 6 argumentos.
440 if (ubicacion != null && estado != null) {
441 subirDatosAFirebase(o3_ppm, temperatura_c, co2_ppm, bat_porc, ubicacion, estado);
442 }
443 // Fin lógica
444
445 }

449 //--- Alertas sobre medidas -----------------------------------------------------------------------
450 /**
451 * @brief Compara las mediciones recibidas con umbrales predefinidos y genera alertas (notificaciones y LiveData).
452 * (co2:int, ozono:float, temperatura:float, bateria:int) -> checkAlerts() -> ()
453 *
454 * - CO2 >= 1200 ppm -> Alerta.
455 * - O3 >= 0.9 ppm -> Alerta.
456 * - Temp > 35ºC -> Alerta.
457 * - Bat <= 15% -> Notificación de Batería Baja.
458 *
459 * @param co2 Concentración de CO2 en ppm.
460 * @param ozono Concentración de Ozono en ppm.
461 * @param temperatura Temperatura en grados Celsius.
462 * @param bateria Porcentaje de batería.
463 */
464 private void checkAlerts(int co2, float ozono, float temperatura, int bateria) {
465 List<String> currentAlertMessages = new ArrayList<>();
466 String currentTime = new SimpleDateFormat("HH:mm", Locale.getDefault()).format(new Date());
467
468 // SI las medidas entran en un rango determinado se mostrara una alerta en el tablón de alertas y en notificaciones
469 if (co2 >= 1200) {
470 String message = currentTime + " - Nivel de CO2 elevado: " + co2 + " ppm";
471 currentAlertMessages.add(message);
472 sendAlertNotification("Alerta de CO2", "Nivel de CO2 elevado: " + co2 + " ppm", CO2_ALERT_ID);
473 } else {
474 cancelAlertNotification(CO2_ALERT_ID);
475 }
476
477 if (ozono >= 0.9) {
478 String message = currentTime + " - Nivel de Ozono elevado: " + String.format(Locale.getDefault(), "%.3f ppm", ozono);
479 currentAlertMessages.add(message);
480 sendAlertNotification("Alerta de Ozono", "Nivel de Ozono elevado: " + String.format(Locale.getDefault(), "%.3f ppm", ozono), OZONE_ALERT_ID);
481 } else {
482 cancelAlertNotification(OZONE_ALERT_ID);
483 }
484
485 if (temperatura > 35) {
486 String message = currentTime + " - Temperatura elevada: " + String.format(Locale.getDefault(), "%.1f ºC", temperatura);
487 currentAlertMessages.add(message);
488 sendAlertNotification("Alerta de Temperatura", "Temperatura elevada: " + String.format(Locale.getDefault(), "%.1f ºC", temperatura), TEMP_ALERT_ID);
489 } else {
490 cancelAlertNotification(TEMP_ALERT_ID);
491 }
492
493 // Si la bateria es baja solo se muestra la notificacion, no la alerta
494 if (bateria <= 15) {
495 sendAlertNotification("Alerta de Batería", "Nivel de batería bajo: " + bateria + "%", BATTERY_ALERT_ID);
496 } else {
497 cancelAlertNotification(BATTERY_ALERT_ID);
498 }
499
500 // Si no hay alertas ni incidencias mostramos: Sin alertas/incidencias
501 if (currentAlertMessages.isEmpty()) {
502 dataHolder.alertData.postValue("Sin alertas");
503 } else {
504 StringJoiner joiner = new StringJoiner("\n\n");
505 for (String msg : currentAlertMessages) {
506 joiner.add(msg);
507 }
508 dataHolder.alertData.postValue(joiner.toString());
509 }
510
511 if ("Conectado".equals(dataHolder.estadoData.getValue())) {
512 dataHolder.incidenciaData.postValue("Sin incidencias");
513 }
514 }

634 /**
635 * @brief Sube las mediciones y el estado de contexto (ubicación, conexión) a Firebase Firestore.
636 * Realiza dos operaciones: guardar un registro en la colección 'mediciones' (historial) y actualizar los campos directos del documento del sensor (última lectura).
637 * (o3_ppm:Float, temp_c:Float, co2_ppm:Integer, bat_porc:Integer, ubicacion:String, estado:String) -> subirDatosAFirebase() -> ()
638 *
639 * Realiza una **doble escritura**:
640 * 1. **Histórico:** Añade un nuevo documento a la subcolección `mediciones`.
641 * 2. **Estado Actual:** Actualiza los campos del documento principal del sensor con `SetOptions.merge()`.
642 *
643 * @param o3_ppm Concentración de Ozono.
644 * @param temp_c Temperatura.
645 * @param co2_ppm Concentración de CO2.
646 * @param bat_porc Porcentaje de batería.
647 * @param ubicacion Dirección legible de la ubicación.
648 * @param estado Estado de conexión del sensor.
649 */
650 public void subirDatosAFirebase(Float o3_ppm, Float temp_c, Integer co2_ppm, Integer bat_porc, String ubicacion, String estado) {
651
652 // Verficar que esten todos los datos necesarios para subir a la bbdd
653 if (o3_ppm == null || temp_c == null || co2_ppm == null || bat_porc == null || ubicacion == null || estado == null) {
654 Log.w("Firestore", "Intento de subida fallido: LiveData incompleto.");
655 return;
656 }
657
658 LecturaSensor nuevaLectura = new LecturaSensor(o3_ppm, temp_c, co2_ppm, bat_porc, ubicacion, estado);
659
660 sensorDocRef.collection("mediciones").add(nuevaLectura) // Se sube el objeto completo
661 .addOnSuccessListener(documentReference -> {
662 Log.d("Firestore", "Historial guardado.");
663 })
664 .addOnFailureListener(e -> {
665 Log.e("Firestore", "Error al escribir en el historial", e);
666 });
667
668
669 // Subida a Campos Directos (Última Lectura)
670
671 Map<String, Object> camposDirectos = new HashMap<>();
672
673 // Datos de la Lectura
674 camposDirectos.put("ozono", o3_ppm);
675 camposDirectos.put("temperatura", temp_c);
676 camposDirectos.put("co2", co2_ppm);
677 camposDirectos.put("bateria", bat_porc);
678
679 // Metadatos (Campos añadidos)
680 camposDirectos.put("ubicacion", ubicacion);
681 camposDirectos.put("estado", estado);
682 camposDirectos.put("ultima_conexion", FieldValue.serverTimestamp());
683
684 // Usamos SET con MERGE
685 sensorDocRef.set(camposDirectos, SetOptions.merge())
686 .addOnSuccessListener(aVoid -> Log.d("Firestore", "Campos directos del sensor actualizados."))
687 .addOnFailureListener(e -> {
688 Log.e("Firestore", "Error al actualizar campos directos: " + e.getMessage());
689 });
690 }
