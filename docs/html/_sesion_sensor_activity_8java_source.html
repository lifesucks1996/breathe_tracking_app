247 /**
248 * @brief Configura los observadores LiveData para actualizar la UI en tiempo real.
249 *
250 * () -> setupObservers() -> ()
251 *
252 * Define la lógica de colores y umbrales para los indicadores visuales:
253 * - **Batería:** Rojo si <= 15%.
254 * - **Ozono (ppm):**
255 * - Verde: < 0.6
256 * - Naranja: 0.6 - 0.9
257 * - Rojo: >= 0.9
258 * - **Temperatura (ºC):**
259 * - Azul: <= 20
260 * - Naranja: 20 - 28
261 * - Rojo: > 28
262 * - **CO2 (ppm):**
263 * - Verde: < 800
264 * - Naranja: 800 - 1200
265 * - Rojo: >= 1200
266 * - **RSSI:** Actualiza el icono de barras según la potencia (dBm).
267 */
268 private void setupObservers() {
269
270 // Ubicacion del dispositivo -----------------------------------
271 dataHolder.locationData.observe(this, address -> {
272 if (address != null) ubicacionTextView.setText(address);
273 });
274
275 // Hora de recepcion de datos del sensor -----------------------
276 dataHolder.timeData.observe(this, time -> {
277 if (time != null) ultimaConexionTextView.setText(time);
278 });
279
280 // Bateria del dispositivo -------------------------------------
281 // Si es baja cambiamos el color a rojo para que destaque, si no el texto se ve en negro
282 dataHolder.bateriaData.observe(this, bateria -> {
283 if (bateria != null) {
284 bateriaTextView.setText(String.format(Locale.getDefault(), "%d%%", bateria));
285 if (bateria <= 15) {
286 bateriaTextView.setTextColor(ContextCompat.getColor(this, R.color.progress_red));
287 } else {
288 bateriaTextView.setTextColor(Color.BLACK);
289 }
290 }
291 });
292
293 // Dato de ozono -----------------------------------------------
294 // Mostramos el dato de ozono y ademas en la progress bar se muestra por rangos de colores:
295 // Verde: 0-0.6 ppm, Amarillo: 0.6-0.9 ppm y Rojo: >0.9 ppm
296 dataHolder.ozonoData.observe(this, ozono -> {
297 if (ozono != null) {
298 ozonoTextView.setText(String.format(Locale.getDefault(), "%.3f ppm", ozono));
299 ozonoProgressBar.setProgress((int) (ozono * 1000));
300 Drawable d = (ozono < 0.6) ? ContextCompat.getDrawable(this, R.drawable.progress_bar_green) : (ozono < 0.9) ? ContextCompat.getDrawable(this, R.drawable.progress_bar_orange) : ContextCompat.getDrawable(this, R.drawable.progress_bar_red);
301 ozonoProgressBar.setProgressDrawable(d);
302 }
303
304 });
305
306 // Dato de temperatura ----------------------------------------
307 // Mostramos el dato de temperatura y ademas en la progress bar se muestra por rangos de colores:
308 // Verde: 0-20 ºC, Amarillo: 20-28 ºC y Rojo: >28 ºC
309 dataHolder.temperaturaData.observe(this, temperatura -> {
310 if (temperatura != null) {
311 temperaturaTextView.setText(String.format(Locale.getDefault(), "%.1f ºC", temperatura));
312 temperaturaProgressBar.setProgress(temperatura.intValue());
313 Drawable d = (temperatura <= 20) ? ContextCompat.getDrawable(this, R.drawable.progress_bar_blue) : (temperatura <= 28) ? ContextCompat.getDrawable(this, R.drawable.progress_bar_orange) : ContextCompat.getDrawable(this, R.drawable.progress_bar_red);
314 temperaturaProgressBar.setProgressDrawable(d);
315 }
316 });
317
318 // Dato de Co2 ------------------------------------------------
319 // Mostramos el dato de Co2 y ademas en la progress bar se muestra por rangos de colores:
320 // Verde: 0-800 ppm, Amarillo: 800-1200 ppm y Rojo: >1200 ppm
321 dataHolder.co2Data.observe(this, co2 -> {
322 if (co2 != null) {
323 co2TextView.setText(String.format(Locale.getDefault(), "%d ppm", co2));
324 co2ProgressBar.setProgress(co2);
325 Drawable d = (co2 < 800) ? ContextCompat.getDrawable(this, R.drawable.progress_bar_green) : (co2 < 1200) ? ContextCompat.getDrawable(this, R.drawable.progress_bar_orange) : ContextCompat.getDrawable(this, R.drawable.progress_bar_red);
326 co2ProgressBar.setProgressDrawable(d);
327 }
328 });
329
330 // RSSI (Señal) ------------------------------------------------
331 // Actualizamos el icono de la señal según el valor RSSI (dBm)
332 dataHolder.rssiData.observe(this, rssi -> {
333 if (rssi != null) {
334 if (rssi >= -60) {
335 imgSignal.setImageResource(R.drawable.ic_signal_bars_4); // Excelente
336 } else if (rssi >= -70) {
337 imgSignal.setImageResource(R.drawable.ic_signal_bars_3); // Buena
338 } else if (rssi >= -80) {
339 imgSignal.setImageResource(R.drawable.ic_signal_bars_2); // Regular
340 } else if (rssi >= -90) {
341 imgSignal.setImageResource(R.drawable.ic_signal_bars_1); // Mala
342 } else {
343 imgSignal.setImageResource(R.drawable.ic_signal_bars_0); // Muy mala/Sin señal
344 }
345 }
346 });
347
348 // Alertas ----------------------------------------------------
349 dataHolder.alertData.observe(this, alert -> {
350 if(alert != null) alertaTextView.setText(alert); // si no es null muestra el texto de alerta
351
352 });
353
354 // Incidencias ------------------------------------------------
355 dataHolder.incidenciaData.observe(this, incidencia -> {
356 if(incidencia != null) incidenciaTextView.setText(incidencia); // si no es null muestra el texto de incidencia
357 });
358
359 // Estado conectado - desconectado ----------------------------
360 // Si esta conectado cambiamos el color a verde, si no el texto se ve en rojo
361 dataHolder.estadoData.observe(this, estado -> {
362 if (estado != null) {
363 estadoTextView.setText(estado);
364 if ("Conectado".equals(estado)) {
365 estadoTextView.setTextColor(ContextCompat.getColor(this, R.color.progress_green));
366 reportarIncidenciaTextView.setVisibility(View.GONE);
367 } else {
368 estadoTextView.setTextColor(ContextCompat.getColor(this, R.color.progress_red));
369 reportarIncidenciaTextView.setVisibility(View.VISIBLE);
370 imgSignal.setImageResource(R.drawable.ic_signal_bars_0); // Si desconectado, señal 0
371 }
372 }
373 });

382 /**
383 * @brief Verifica si todos los permisos necesarios (Ubicación, Bluetooth) están concedidos.
384 * Si no, solicita al usuario.
385 * () -> checkPermissionsAndStartService() -> ()
386 *
387 * Si faltan permisos, utiliza @ref requestPermissionLauncher para solicitarlos.
388 * Si están concedidos, llama inmediatamente a @ref startTrackingService.
389 *
390 */
391 private void checkPermissionsAndStartService() {
392 String[] permissionsToRequest = {
393 Manifest.permission.ACCESS_FINE_LOCATION,
394 Manifest.permission.BLUETOOTH_SCAN,
395 Manifest.permission.BLUETOOTH_CONNECT
396 };
397
398 boolean allPermissionsGranted = true;
399 for (String permission : permissionsToRequest) {
400 if (ContextCompat.checkSelfPermission(this, permission) != PackageManager.PERMISSION_GRANTED) {
401 allPermissionsGranted = false;
402 break;
403 }
404 }
405
406 //Si hay permisos empezamos el servicio de tracking
407 if (allPermissionsGranted) {
408 startTrackingService();
409 } else {
410 requestPermissionLauncher.launch(permissionsToRequest);
411 }
412 }
