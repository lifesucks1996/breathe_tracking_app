81 /**
82 * @brief Lógica de conexión y envío en hilo secundario.
83 *
84 * 1. Configura las propiedades (`Properties`) para TLS/SSL en el puerto 465 de Gmail.
85 * 2. Inicia sesión (`Session`) con las credenciales del robot.
86 * 3. Crea el objeto `MimeMessage`.
87 * 4. Asigna un alias ("⚠️ Alertas Breathe Tracking") al remitente.
88 * 5. Genera el contenido HTML llamando a @ref construirHTML.
89 * 6. Envía el mensaje mediante `Transport.send()`.
90 *
91 * @param params Void (no se usan parámetros variables).
92 * @return null
93 */
94 @Override
95 protected Void doInBackground(Void... params) {
96 Properties props = new Properties();
97 // Configuración para conectarse a Gmail
98 props.put("mail.smtp.host", "smtp.gmail.com");
99 props.put("mail.smtp.socketFactory.port", "465");
100 props.put("mail.smtp.socketFactory.class", "javax.net.ssl.SSLSocketFactory");
101 props.put("mail.smtp.auth", "true");
102 props.put("mail.smtp.port", "465");
103
104 // Iniciar sesión
105 session = Session.getDefaultInstance(props, new javax.mail.Authenticator() {
106 protected PasswordAuthentication getPasswordAuthentication() {
107 return new PasswordAuthentication(EMAIL_ROBOT, PASSWORD_ROBOT);
108 }
109 });
110
111 try {
112 MimeMessage mm = new MimeMessage(session);
113
114 /** @brief Configuración del remitente con alias visual personalizado para apariencia profesional. */
115 mm.setFrom(new InternetAddress(EMAIL_ROBOT, "⚠️ Alertas Breathe Tracking"));
116 // ---------------------------
117
118 mm.addRecipient(Message.RecipientType.TO, new InternetAddress(emailDestino));
119 mm.setSubject(asunto);
120
121 String contenidoHTML = construirHTML(asunto, mensaje);
122 /** @brief Establece el contenido como HTML con codificación UTF-8 para soporte de tildes. */
123 mm.setContent(contenidoHTML, "text/html; charset=utf-8");
124
125 Transport.send(mm);
126
127 } catch (Exception e) {
128 e.printStackTrace();
129 }
130 return null;
131 }
