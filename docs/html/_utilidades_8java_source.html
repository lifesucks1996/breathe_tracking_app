41 * @brief Convierte una cadena de 16 caracteres en un objeto UUID.
42 * (uuid:String) -> stringToUUID() -> UUID
43 *
44 * @details
45 * * La cadena se divide en dos partes de 8 caracteres. Cada parte se convierte

47 *
48 * @param uuid La cadena de 16 caracteres a convertir.
49 * @return El objeto UUID resultante.
50 * @throws Error Si la cadena no tiene exactamente 16 caracteres.
51 */
52 public static UUID stringToUUID( String uuid ) {
53 if ( uuid.length() != 16 ) {
54 throw new Error( "stringUUID: string no tiene 16 caracteres ");
55 }
56 byte[] comoBytes = uuid.getBytes();
57
58 String masSignificativo = uuid.substring(0, 8);
59 String menosSignificativo = uuid.substring(8, 16);
60 UUID res = new UUID( Utilidades.bytesToLong( masSignificativo.getBytes() ), Utilidades.bytesToLong( menosSignificativo.getBytes() ) );

154 /**
155 * @brief Convierte un array de bytes en un valor entero (int) con lógica de desplazamiento de bits y manejo de signo (complemento a 2).
156 * (bytes:byte[]) -> bytesToIntOK() -> int
157 *
158 * @details
159 * Realiza un desplazamiento de bits (Bit shifting) para reconstruir el entero.
160 * Incluye una lógica manual para manejar el complemento a 2 si se detecta un bit de signo específico.
161 *
162 *
163 *
164 * @note Este método implementa una conversión byte a byte con desplazamiento (Big Endian)
165 * y una lógica de manejo de signo que parece ser una adaptación manual para enteros con signo.
166 * @param bytes El array de bytes de entrada (máximo 4 bytes).
167 * @return El valor int resultante.
168 * @throws Error Si el array de bytes supera los 4 bytes.
169 */
170 public static int bytesToIntOK( byte[] bytes ) {
171 if (bytes == null ) {
172 return 0;
173 }
174
175 if ( bytes.length > 4 ) {
176 throw new Error( "demasiados bytes para pasar a int ");
177 }
178 int res = 0;
179
180
181
182 for( byte b : bytes ) {
183 res = (res << 8) // * 16
184 + (b & 0xFF); // para quedarse con 1 byte (2 cuartetos) de lo que haya en b
185 } // for
186
187 if ( (bytes[ 0 ] & 0x8) != 0 ) {
188 // si tiene signo negativo (un 1 a la izquierda del primer byte
189 res = -(~(byte)res)-1; // complemento a 2 (~) de res pero como byte, -1
190 }
191 return res;
192 } // ()
